const PLANILHA_ID = '1mked_mHJTp0kYX6A9RSBVwVo_Tc4xfWqighed76yo1Q'; const NOME_ABA = 'Registros'; function obterPlanilha() { return SpreadsheetApp.openById(PLANILHA_ID); } /* ============================== CONFIGURAR PLANILHA ============================== */ function configurarPlanilha() { const planilha = obterPlanilha(); let aba = planilha.getSheetByName(NOME_ABA); if (!aba) { aba = planilha.insertSheet(NOME_ABA); } if (aba.getLastRow() === 0) { aba.appendRow([ 'Data', 'Motorista', 'Hora de Saída', 'Veículo', 'Placa', 'Destino', 'KM Saída', 'KM Chegada', 'Hora de Chegada', 'KM Percorrido', 'Assinatura' ]); } return aba; } /* ============================== SALVAR SAÍDA ============================== */ function salvarSaida(dados) { if (!dados.motorista) { throw new Error('Informe o motorista.'); } if (!dados.horaSaida) { throw new Error('Informe a hora de saída.'); } if (!dados.veiculo) { throw new Error('Informe o veículo.'); } if (!dados.placa) { throw new Error('Informe a placa.'); } if (!dados.destino) { throw new Error('Informe o destino.'); } if ( dados.kmSaida === '' || dados.kmSaida === undefined ) { throw new Error('Informe o KM de saída.'); } const kmSaida = Number(dados.kmSaida); if (isNaN(kmSaida)) { throw new Error('O KM de saída deve ser um número.'); } const aba = configurarPlanilha(); aba.appendRow([ new Date(), dados.motorista, dados.horaSaida, dados.veiculo, dados.placa, dados.destino, kmSaida, '', '', '', '' ]); return { mensagem: 'KM de saída salvo com sucesso!', linha: aba.getLastRow() }; } /* ============================== SALVAR CHEGADA ============================== */ function salvarChegada(dados) { const aba = obterPlanilha().getSheetByName(NOME_ABA); if (!aba) { throw new Error( 'A aba Registros não existe.' ); } if ( dados.kmChegada === '' || dados.kmChegada === undefined ) { throw new Error( 'Informe o KM de chegada.' ); } const kmChegada = Number(dados.kmChegada); if (isNaN(kmChegada)) { throw new Error( 'O KM de chegada deve ser um número.' ); } /* VERIFICAR ASSINATURA */ if ( !dados.assinatura || dados.assinatura.trim() === '' ) { throw new Error( 'Informe a assinatura do motorista.' ); } const ultimaLinha = aba.getLastRow(); /* Procura a última viagem aberta do mesmo motorista e veículo. */ for ( let linha = ultimaLinha; linha >= 2; linha-- ) { const motorista = aba.getRange(linha, 2).getValue(); const veiculo = aba.getRange(linha, 4).getValue(); const chegada = aba.getRange(linha, 8).getValue(); if ( motorista == dados.motorista && veiculo == dados.veiculo && chegada === '' ) { const kmSaida = Number( aba.getRange(linha, 7).getValue() ); if (kmChegada < kmSaida) { throw new Error( 'O KM de chegada não pode ser menor que o KM de saída.' ); } const kmPercorrido = kmChegada - kmSaida; /* HORA DE CHEGADA */ let horaChegada = dados.horaChegada; /* Se o aplicativo não enviar a hora, o próprio Apps Script registra a hora atual. */ if (!horaChegada) { horaChegada = Utilities.formatDate( new Date(), Session.getScriptTimeZone(), 'HH:mm' ); } /* ATUALIZA A LINHA */ aba.getRange(linha, 8) .setValue(kmChegada); aba.getRange(linha, 9) .setValue(horaChegada); aba.getRange(linha, 10) .setValue(kmPercorrido); aba.getRange(linha, 11) .setValue(dados.assinatura); return 'Viagem encerrada com sucesso!'; } } throw new Error( 'Não foi encontrada uma viagem em aberto.' ); } /* ============================== RECEBER DADOS DO APLICATIVO ============================== */ function doPost(e) { try { const dados = JSON.parse( e.postData.contents ); let resultado; if (dados.acao === 'saida') { resultado = salvarSaida(dados); } else if (dados.acao === 'chegada') { resultado = salvarChegada(dados); } else { throw new Error( 'Ação não reconhecida.' ); } return ContentService .createTextOutput( JSON.stringify({ sucesso: true, mensagem: resultado.mensagem || resultado, linha: resultado.linha || null }) ) .setMimeType( ContentService.MimeType.JSON ); } catch (erro) { return ContentService .createTextOutput( JSON.stringify({ sucesso: false, mensagem: erro.message }) ) .setMimeType( ContentService.MimeType.JSON ); } } /* ============================== TESTE ============================== */ function doGet() { return ContentService .createTextOutput( 'Controle de Frota COREN-PB funcionando!' );
