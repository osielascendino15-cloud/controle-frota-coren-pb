/* ASSINATURA PARA CELULAR E COMPUTADOR */ function prepararCanvas() { const proporcao = window.devicePixelRatio || 1; const largura = canvas.clientWidth; const altura = canvas.clientHeight; canvas.width = largura * proporcao; canvas.height = altura * proporcao; ctx.setTransform(1, 0, 0, 1, 0, 0); ctx.scale(proporcao, proporcao); ctx.lineWidth = 3; ctx.lineCap = "round"; ctx.lineJoin = "round"; } function obterPosicao(evento) { const retangulo = canvas.getBoundingClientRect(); return { x: evento.clientX - retangulo.left, y: evento.clientY - retangulo.top }; } function iniciarAssinatura(evento) { evento.preventDefault(); desenhando = true; assinaturaTemConteudo = true; const posicao = obterPosicao(evento); ctx.beginPath(); ctx.moveTo( posicao.x, posicao.y ); } function desenharAssinatura(evento) { if (!desenhando) return; evento.preventDefault(); const posicao = obterPosicao(evento); ctx.lineTo( posicao.x, posicao.y ); ctx.stroke(); } function finalizarAssinatura(evento) { if (evento) { evento.preventDefault(); } desenhando = false; } /* EVENTOS PARA DEDO, CANETA E MOUSE */ canvas.addEventListener( "pointerdown", iniciarAssinatura, { passive: false } ); canvas.addEventListener( "pointermove", desenharAssinatura, { passive: false } ); canvas.addEventListener( "pointerup", finalizarAssinatura, { passive: false } ); canvas.addEventListener( "pointercancel", finalizarAssinatura, { passive: false } ); canvas.addEventListener( "pointerleave", finalizarAssinatura, { passive: false } ); /* EVENTOS EXTRAS PARA CELULARES */ canvas.addEventListener( "touchstart", function(evento) { const toque = evento.touches[0]; iniciarAssinatura({ clientX: toque.clientX, clientY: toque.clientY, preventDefault: function() { evento.preventDefault(); } }); }, { passive: false } ); canvas.addEventListener( "touchmove", function(evento) { if (!desenhando) return; const toque = evento.touches[0]; desenharAssinatura({ clientX: toque.clientX, clientY: toque.clientY, preventDefault: function() { evento.preventDefault(); } }); }, { passive: false } ); canvas.addEventListener( "touchend", function(evento) { finalizarAssinatura({ preventDefault: function() { evento.preventDefault(); } }); }, { passive: false } );
